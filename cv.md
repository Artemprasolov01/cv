## Code Example

```cpp
// A simple program to find the maximum element in an array
#include <iostream>
#include <vector>
using namespace std;

int findMax(const vector<int>& arr) {
    if (arr.empty()) {
        throw runtime_error("Array is empty");
    }
    
    int maxVal = arr[0];
    for (int num : arr) {
        if (num > maxVal) {
            maxVal = num;
        }
    }
    return maxVal;
}

int main() {
    vector<int> numbers = {5, 12, 3, 8, 20, 7};
    cout << "Maximum element: " << findMax(numbers) << endl;
    return 0;
}
```
